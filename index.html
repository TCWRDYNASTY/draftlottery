/* ========================================================= RACE POSITION SETTINGS ========================================================= */ /* The finish point is the visual 10-yard line on racefield.png. Keeping this in one place makes the runner position, reveal position, and saved draft display all use the exact same finish point. */ const START_POSITION = 8; const FINISH_POSITION = 84; /* ========================================================= START RACE ========================================================= */ async function start(){ if( raceRunning ){ return false; } resetVisuals(); initializeDraftOrderDisplay(); let officialOrder = savedDraftOrder && savedDraftOrder.length === teams.length ? savedDraftOrder.map( item => ({ pick:item.pick, team:{ ...item.team } }) ) : []; if( officialOrder.length !== teams.length ){ showStatus( "Creating the official draft order..." ); officialOrder = createDraftOrder(); if( officialOrder.length !== teams.length ){ showStatus( "Could not create a complete draft order." ); return false; } savedDraftOrder = officialOrder.map( item => ({ pick: item.pick, team:{ ...item.team } }) ); const saved = await saveDraftOrderToSupabase( officialOrder ); if( !saved ){ console.warn( "Supabase could not save the draft. Continuing with the locally generated order for this race." ); showStatus( "Draft order created — race starting." ); } } const order = officialOrder.map( item => ({ pick: item.pick, team:{ ...item.team } }) ); raceRunning = true; showStatus( "The draft is locked in — the race is on!" ); runners = teams.map( ( team, index ) => ({ team:{ ...team }, el: addRunner( team, index ), finish:0, revealed:false, rank:-1 }) ); /* 60-SECOND RACE TIMING The final runner reaches the finish line at exactly 60 seconds. */ const raceDuration = 60000; const baseFinish = 50000; const finishGap = teams.length > 1 ? (raceDuration - baseFinish) / (teams.length - 1) : 0; order.forEach( ( item, rank ) => { const runner = runners.find( r => String(r.team.id) === String(item.team.id) ); if( !runner ){ return; } runner.pick = item.pick; runner.rank = rank; runner.finish = baseFinish + rank * finishGap; /* Each runner gets a different race personality. Some start hot. Some fade. Some get a mid-race boost. Some make a late surge. */ runner.behavior = { earlyBoost: (Math.random() * .075) + .025, earlyFade: -Math.random() * .075, midBoost: (Math.random() * .12) - .055, lateBoost: (Math.random() * .10) - .045, waveAmount: .012 + Math.random() * .022, waveSpeed: 2.4 + Math.random() * 2.2, wavePhase: Math.random() * Math.PI * 2 }; } ); const startTime = performance.now(); function revealRunner( runner ){ if( runner.revealed ){ return; } runner.revealed = true; runner.el.classList.add( "finished" ); /* EXACT FINISH POSITION This is the 10-yard-line position. */ runner.el.style.left = FINISH_POSITION + "%"; const result = order.find( item => String(item.team.id) === String(runner.team.id) ); if( result ){ revealDraftResult( result ); } } function frame( now ){ const elapsed = now - startTime; const overallProgress = Math.min( 1, elapsed / raceDuration ); runners.forEach( runner => { const finishTime = runner.finish; const individualProgress = Math.min( 1, elapsed / finishTime ); const sharedProgress = overallProgress; /* MORE SEPARATION The old animation used 88% shared movement and only 12% individual movement. This version gives runners much more independent movement. */ const behavior = runner.behavior || { earlyBoost:0, earlyFade:0, midBoost:0, lateBoost:0, waveAmount:0, waveSpeed:3, wavePhase:0 }; const progress = Math.min( 1, ( sharedProgress * .52 ) + ( individualProgress * .48 ) ); /* HOT START */ const earlyWindow = Math.max( 0, 1 - Math.abs(progress - .14) / .18 ); /* FADE */ const fadeWindow = Math.max( 0, 1 - Math.abs(progress - .34) / .20 ); /* MID-RACE BOOST */ const midWindow = Math.max( 0, 1 - Math.abs(progress - .58) / .20 ); /* LATE SURGE */ const lateWindow = Math.max( 0, 1 - Math.abs(progress - .82) / .18 ); /* SMALL CONTINUOUS PACE VARIATION */ const wave = Math.sin( progress * Math.PI * behavior.waveSpeed + behavior.wavePhase ) * behavior.waveAmount; const movementOffset = ( behavior.earlyBoost * earlyWindow ) + ( behavior.earlyFade * fadeWindow ) + ( behavior.midBoost * midWindow ) + ( behavior.lateBoost * lateWindow ) + wave; const startPosition = START_POSITION; const basePosition = startPosition + progress * ( FINISH_POSITION - startPosition ); /* Never allow a visual surge to cross the finish line before the runner's official scheduled finish time. When the finish time arrives, the runner moves exactly to the 10-yard line and the pick is revealed. */ const currentProgress = elapsed >= finishTime ? 1 : Math.min( .985, Math.max( 0, progress + movementOffset ) ); const currentPosition = startPosition + currentProgress * ( FINISH_POSITION - startPosition ); runner.el.style.left = `${currentPosition}%`; if( elapsed >= finishTime ){ revealRunner( runner ); } } ); if( runners.every( runner => runner.revealed ) ){ raceRunning = false; renderResults( [...savedDraftOrder] ); showStatus( "DRAFT ORDER COMPLETE — Pick #1 is the winner." ); return; } raf = requestAnimationFrame( frame ); } raf = requestAnimationFrame( frame ); return true; }
